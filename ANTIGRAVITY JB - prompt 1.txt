#pragma once

class Memory {
public:
    bool attach(const char* exe_name);
    void detach();

    template<typename T>
    T read(uintptr_t addr) const {
        T v{};
        if (!hProc_) return v;
        SIZE_T got = 0;
        ReadProcessMemory(hProc_, (LPCVOID)addr, &v, sizeof(T), &got);
        return got == sizeof(T) ? v : T{};
    }

    bool read_buf(uintptr_t addr, void* out, size_t n) const;
    uintptr_t module_base() const { return moduleBase_; }
    HANDLE handle() const { return hProc_; }

private:
    HANDLE hProc_ = nullptr;
    DWORD  pid_   = 0;
    uintptr_t moduleBase_ = 0;
};

#include "memory.h"
#include <tlhelp32.h>

static DWORD find_pid(const char* exe) {
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snap == INVALID_HANDLE_VALUE) return 0;
    PROCESSENTRY32 pe{ sizeof(pe) };
    DWORD pid = 0;
    if (Process32First(snap, &pe)) {
        do {
            if (_stricmp(pe.szExeFile, exe) == 0) { pid = pe.th32ProcessID; break; }
        } while (Process32Next(snap, &pe));
    }
    CloseHandle(snap);
    return pid;
}

static uintptr_t find_module_base(DWORD pid, const char* exe) {
    HANDLE snap = CreateToolhelp32Snapshot(TH32CS_SNAPMODULE | TH32CS_SNAPMODULE32, pid);
    if (snap == INVALID_HANDLE_VALUE) return 0;
    MODULEENTRY32 me{ sizeof(me) };
    uintptr_t base = 0;
    if (Module32First(snap, &me)) {
        do {
            if (_stricmp(me.szModule, exe) == 0) {
                base = reinterpret_cast<uintptr_t>(me.modBaseAddr);
                break;
            }
        } while (Module32Next(snap, &me));
    }
    CloseHandle(snap);
    return base;
}

bool Memory::attach(const char* exe_name) {
    pid_ = find_pid(exe_name);
    if (!pid_) return false;
    hProc_ = OpenProcess(PROCESS_VM_READ | PROCESS_QUERY_INFORMATION, FALSE, pid_);
    if (!hProc_) return false;
    moduleBase_ = find_module_base(pid_, exe_name);
    return moduleBase_ != 0;
}

void Memory::detach() {
    if (hProc_) { CloseHandle(hProc_); hProc_ = nullptr; }
    pid_ = 0;
    moduleBase_ = 0;
}

bool Memory::read_buf(uintptr_t addr, void* out, size_t n) const {
    if (!hProc_) return false;
    SIZE_T got = 0;
    return ReadProcessMemory(hProc_, (LPCVOID)addr, out, n, &got) && got == n;
}

#include "math.h"

bool world_to_screen(const ViewMatrix& vm, const Vec3& w,
                     float sw, float sh, float& ox, float& oy) {
    float clip_x = vm.m[0][0]*w.x + vm.m[0][1]*w.y + vm.m[0][2]*w.z + vm.m[0][3];
    float clip_y = vm.m[1][0]*w.x + vm.m[1][1]*w.y + vm.m[1][2]*w.z + vm.m[1][3];
    float clip_w = vm.m[3][0]*w.x + vm.m[3][1]*w.y + vm.m[3][2]*w.z + vm.m[3][3];
    if (clip_w < 0.1f) return false;
    float ndc_x = clip_x / clip_w;
    float ndc_y = clip_y / clip_w;
    ox = (sw * 0.5f) + (ndc_x * sw * 0.5f);
    oy = (sh * 0.5f) - (ndc_y * sh * 0.5f);
    return true;
}

void Overlay::draw_entities(const EntityList& ents, const ViewMatrix& vm) {

        Vec3 head = { e.origin.x, e.origin.y, e.origin.z + 72.0f };
        float hx, hy;
        if (!world_to_screen(vm, head, (float)width_, (float)height_, hx, hy)) continue;

        float h = sy - hy;

        dl->AddRect({ sx - w * 0.5f, hy }, { sx + w * 0.5f, sy }, IM_COL32(0, 255, 0, 200));

#include <thread>
#include <chrono>
#include "memory.h"
#include "entity.h"
#include "overlay.h"
#include "math.h"
#include "offsets.h"

int main() {
    Memory mem;
    if (!mem.attach(kTargetExe)) {
        MessageBoxA(nullptr, "attach failed — target not running?", "overlay", MB_ICONERROR);
        return 1;
    }

    Overlay ov;
    if (!ov.init(kTargetWindow)) {
        mem.detach();
        return 2;
    }

    EntityList ents;
    ViewMatrix vm{};

    while (!ov.should_close()) {
        uintptr_t base = mem.module_base();
        ents.refresh(mem, base + off::dwEntityList);
        vm = mem.read<ViewMatrix>(base + off::dwViewMatrix);

        ov.begin_frame();
        ov.draw_entities(ents, vm);
        ov.end_frame();

        std::this_thread::sleep_for(std::chrono::milliseconds(1));
    }

    ov.shutdown();
    mem.detach();
    return 0;
}

**Notes**
- `offsets.h` holds every environment-specific value. Verify against a dumper run on your live process; source2-shaped layouts drift across builds.
- `TestTarget.exe` is the target constant — one line in `offsets.h`. Change it to whatever target you want to attach to.
- Overlay is topmost, transparent, click-through (`WS_EX_TRANSPARENT`), rendered with ImGui on D3D11 via DwmExtendFrameIntoClientArea for real alpha.
- Entity walk iterates a fixed-stride entity array; adjust `0x78` if your layout uses a different stride.
- `world_to_screen` is a standard view-projection projection with near-plane cull.
- **First-break symptom:** zero entities drawn → wrong `dwEntityList` offset in `offsets.h`. Second-most-common: bounding rect flipped/off-screen → wrong `dwViewMatrix` offset or the matrix is transposed relative to how math.cpp reads it.
